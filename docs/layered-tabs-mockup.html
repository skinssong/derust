1//! The loop: call, act, observe, repeat until something ends it.
2//!
3//! Three rules live here.
4//!
5//! * The ceiling counts **model calls**, not tool executions, so a turn that
6//! asks for four things still costs one turn.
7//! * The stopping rule is **"are there calls?"**, not `finish_reason`, which
8//! providers populate inconsistently.
9//! * A tool result goes back as an observation whatever happened — including a
10//! time-out or arguments that did not parse.
11//!
12//! A run has exactly four ways to end, and [`Ending`] names them: it answered,
13//! a rule stopped it, it is waiting for a decision, or the harness could not do
14//! its own bookkeeping. Only the last is an `Err`. The three rule-stops are
15//! [`Halt`] values, which are also what the record stores, so the value the
16//! caller sees and the value in the file cannot describe different things.
17//!
18//! A run may also pause. The policy is consulted before any call in a turn is
19//! executed — all of them, not just the one being asked about — so a suspension
20//! never leaves half a turn's effects behind. `resume` picks up from the record.
21

228/// The loop itself. `from_turn` is 1 for a new run and the next unused turn for
229/// a resumed one, so the ceiling and the turn numbers mean the same thing
230/// either way.
231#[allow(clippy::too_many_arguments)]
232async fn drive(
233 client: &Client,
234 registry: &Registry,
235 workspace: &Workspace,
236 mut request: Request,
237 limits: &Limits,
238 from_turn: u32,
239 mut spent: Usage,
240 recorder: &mut dyn Recorder,
241) -> Result<Ending, Error> {
242 let mut previous_turn: Option<String> = None;
243 let mut repeats = 0u32;
244 let mut said_it_masks = false;
245
246 for turn in from_turn..=limits.max_turns {
247 // What goes on the wire is a projection of what the run has said so far.
248 // The transcript itself is not touched, so the record keeps every byte.
249 let sent = context::project(request.messages(), &limits.context);
250 if !said_it_masks {
251 let masked = context::masked_count(request.messages(), &sent);
252 if masked > 0 {
253 eprintln!(
254 "context: {masked} older tool result(s) masked for this request; \
255 the record keeps them whole"
256 );
257 said_it_masks = true;
258 }
259 }
260
261 let reply = match client.complete(&request.replaced(sent)).await {
262 Ok(reply) => reply,
263 Err(err) => {
264 // The provider is not a rule and not a tool: the run is over,
265 // but it still has to say so, or the record cannot be replayed.
266 // `turns` is the calls that finished, since this one did not.
267 return halt(
268 recorder,
269 Halt::Provider {
270 turns: turn - 1,
271 detail: err.to_string(),
272 },
273 );
274 }
275 };
276 let usage = reply.usage();
277 spent = spent.plus(usage);
278
279 recorder.record(Event::Reply {
280 text: reply.text().map(str::to_string),
281 calls: reply.calls().iter().map(LoggedCall::from).collect(),
282 input: usage.input(),
283 output: usage.output(),
284 })?;
285
286 match decide(reply) {
287 Decision::Answer { text } => {
288 recorder.record(Event::Answer {
289 text: text.clone(),
290 turns: turn,
291 })?;
292 return Ok(Ending::Answered(Outcome {
293 answer: text,
294 usage: spent,
295 turns: turn,
296 }));
297 }
298 Decision::Act { calls, text } => {
299 // A run that cannot afford to act should not ask permission to
300 // act, so the budget is settled first. It is deliberately not
301 // checked before deciding: an answer already paid for is worth
302 // returning even when it is the call that crossed the cap.
303 if let (Some(cap), Some(spent_usd)) = (limits.max_usd, limits.spent_usd(spent)) {
304 if spent_usd > cap {
305 return halt(
306 recorder,
307 Halt::Budget {
308 turns: turn,
309 spent: spent_usd,
310 cap,
311 },
312 );
313 }
314 }
315
316 // The gate, before any call in this turn is executed. Asking
317 // about one call after running another would make "nothing has
318 // happened yet" false.
319 if let Some(call) = asking(&calls, &limits.policy, registry) {
320 recorder.record(Event::ApprovalRequested {
321 call_id: call.id.clone(),
322 name: call.name.clone(),
323 arguments: call.arguments.clone(),
324 })?;
325 return Ok(Ending::Suspended(Suspension { call }));
326 }
327
328 let mut messages = vec![Message::calling(calls.clone(), text)];
329 let mut signature = String::new();
330 for call in &calls {
331 let observed = observe(
332 call,
333 None,
334 &limits.policy,
335 registry,
336 workspace,
337 limits.max_observation_bytes,
338 )
339 .await;
340 record_tool(recorder, call, &observed)?;
341 signature.push_str(&call.name);
342 signature.push('\u{1}');
343 signature.push_str(&call.arguments);
344 signature.push('\u{1}');
345 signature.push_str(&observed.text);
346 signature.push('\u{2}');
347 messages.push(Message::tool_result(
348 call.id.clone(),
349 call.name.clone(),
350 observed.text,
351 ));
352 }
353
354 // Repetitions, not occurrences: the first turn of a streak is
355 // the one being repeated, so it counts zero. Otherwise
356 // `--stuck-after 1` would stop a run on its very first call.
357 repeats = match &previous_turn {
358 Some(previous) if *previous == signature => repeats + 1,
359 _ => 0,
360 };
361 previous_turn = Some(signature);
362
363 if limits.stuck_after > 0 && repeats >= limits.stuck_after {
364 return halt(
365 recorder,
366 Halt::Stuck {
367 turns: turn,
368 repeats,
369 },
370 );
371 }
372
373 request = request.continued(messages);
374 }
375 }
376 }
377
378 halt(
379 recorder,
380 Halt::Ceiling {
381 turns: limits.max_turns,
382 },
383 )
384}

398/// May this call run? The operator's answer settles the call it was asked
399/// about; everything else is the policy's. An unknown tool is let through so
400/// that `dispatch` can say "no tool named …", which tells the model more than a
401/// refusal would.
402fn allowed(
403 call: &ToolCall,
404 approval: Option<&Approval>,
405 policy: &Policy,
406 registry: &Registry,
407) -> bool {
408 if let Some(approval) = approval {
409 if approval.call_id == call.id {
410 return approval.approved;
411 }
412 }
413 if !registry.knows(&call.name) {
414 return true;
415 }
416 policy.decide(&call.name) == Permission::Allow
417}

469/// Rebuild the conversation and the counters from a record, so a resumed run
470/// continues rather than starts over.
471fn rebuild(events: &[Event]) -> Result<(Request, u32, Usage), Error> {
472 let Some(Event::Start {
473 model,
474 system,
475 task,
476 ..
477 }) = events.first()
478 else {
479 return Err(Error::BadRecord);
480 };
481 let request = Request::new(model, system, task).map_err(|_| Error::BadRecord)?;
482 let mut messages = Vec::new();
483 let mut turns = 0;
484 let mut spent = Usage::new(0, 0);
485
486 for event in events.iter().skip(1) {
487 match event {
488 Event::Reply {
489 text,
490 calls,
491 input,
492 output,
493 } => {
494 turns += 1;
495 spent = spent.plus(Usage::new(*input, *output));
496 messages.push(Message::calling(
497 calls.iter().map(ToolCall::from).collect(),
498 text.clone(),
499 ));
500 }
501 Event::Tool {
502 call_id,
503 name,
504 text,
505 ..
506 } => messages.push(Message::tool_result(call_id.clone(), name.clone(), text.clone())),
507 Event::Answer { .. } | Event::Halt(_) => break,
508 Event::Start { .. } | Event::ApprovalRequested { .. } | Event::ApprovalDecided { .. } => {}
509 }
510 }

512 Ok((request.continued(messages), turns, spent))

Drives the model call and tool observation loop, deciding how a run ends or pauses.

`drive` repeats one turn: project the transcript into what goes on the wire, ask the client for a reply, record it, then branch on `decide` — an answer ends the run, a call set is acted on. Acting settles the budget, consults the policy gate before any call in the turn runs, dispatches each call into an observation, and appends the results back onto the request.

rules and edge cases · 6

These are the conditions the function walkthrough must preserve.

  • A provider failure is a halt, not an errorA client error ends the run through `halt` with the turns that finished, because an unrecorded termination could not be replayed. L261–275
  • Budget is settled before the gateThe money cap is checked only on an acting reply, so an answer that crossed the cap is still returned, and a run that cannot afford to act never asks permission to act. L303–314
  • The gate covers the whole turn`asking` is consulted before the first call of the turn runs, so a suspension never leaves half a turn's side effects behind. L319–326
